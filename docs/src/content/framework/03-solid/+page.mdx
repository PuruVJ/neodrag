```sh
npm i @neodrag/core@next @neodrag/solid@next
```

Import built-in plugins from `@neodrag/core/plugins` or `@neodrag/solid/plugins`. Solid uses **`create*`** to attach bindings (`createDraggable`, `createDroppable`, `createSortable`) and **`with*`** to compose plugin lists (`withSortableItemPlugins`). Drop: `@neodrag/solid/drop`. Sortable: `@neodrag/solid/sortable`.

## Usage

Basic usage

```tsx
import { createDraggable } from '@neodrag/solid';

export const App: Component = () => {
  const [draggableRef, setDraggableRef] =
    createSignal<HTMLElement | null>(null);

  createDraggable(draggableRef);

  return <div ref={setDraggableRef}>You can drag me</div>;
};
```

> Note: `@neodrag/solid` uses a shared engine from `@neodrag/core` with one set of delegated pointer listeners for the page, which keeps behavior consistent across frameworks and scales better with many draggables.

With plugins

```tsx
import { createDraggable, axis, grid } from '@neodrag/solid';

export const App: Component = () => {
  const [draggableRef, setDraggableRef] =
    createSignal<HTMLElement | null>(null);

  createDraggable(draggableRef, [axis('x'), grid([10, 10])]);

  return <div ref={setDraggableRef}>Horizontal grid snapping</div>;
};
```

Defining plugins elsewhere with TypeScript

```tsx
import {
  createDraggable,
  axis,
  bounds,
  BoundsFrom,
  type Plugin,
} from '@neodrag/solid';

export const App: Component = () => {
  const [draggableRef, setDraggableRef] =
    createSignal<HTMLElement | null>(null);

  const plugins: Plugin[] = [axis('y'), bounds(BoundsFrom.parent())];
  createDraggable(draggableRef, plugins);

  return <div ref={setDraggableRef}>Type-safe dragging</div>;
};
```

Getting drag state

```tsx
import { createDraggable } from '@neodrag/solid';

export const App: Component = () => {
  const [draggableRef, setDraggableRef] =
    createSignal<HTMLElement | null>(null);
  const dragState = createDraggable(draggableRef);

  createEffect(() => {
    console.log('Position:', dragState().offset);
    console.log('Is dragging:', dragState().isDragging);
  });

  return (
    <div ref={setDraggableRef}>Check console while dragging</div>
  );
};
```

## Reactive Plugins with createCompartment

For dynamic behavior that changes during runtime:

```tsx
import { createSignal } from 'solid-js';
import { createDraggable, axis } from '@neodrag/solid';

export const App: Component = () => {
  const [draggableRef, setDraggableRef] =
    createSignal<HTMLElement | null>(null);
  const [currentAxis, setCurrentAxis] = createSignal<'x' | 'y'>('x');

  createDraggable(draggableRef, [axisCompartment]);

  return (
    <div>
      <div ref={setDraggableRef}>Current axis: {currentAxis()}</div>
      <button
        onClick={() =>
          setCurrentAxis(currentAxis() === 'x' ? 'y' : 'x')
        }
      >
        Switch Axis
      </button>
    </div>
  );
};
```

Multiple reactive compartments

```tsx
import { createSignal } from 'solid-js';
import {
  createDraggable,
  axis,
  bounds,
  BoundsFrom,
  grid,
} from '@neodrag/solid';

export const App: Component = () => {
  const [draggableRef, setDraggableRef] =
    createSignal<HTMLElement | null>(null);
  const [currentAxis, setCurrentAxis] = createSignal<'x' | 'y'>('x');
  const [gridSize, setGridSize] = createSignal(20);
  const [enableBounds, setEnableBounds] = createSignal(false);

  createDraggable(draggableRef, [axisComp, gridComp, boundsComp]);

  return (
    <div>
      <div ref={setDraggableRef}>Reactive draggable</div>

      <div>
        <select
          value={currentAxis()}
          onChange={(e) =>
            setCurrentAxis(e.target.value as 'x' | 'y')
          }
        >
          <option value="x">X</option>
          <option value="y">Y</option>
        </select>

        <input
          type="range"
          min="10"
          max="50"
          value={gridSize()}
          onInput={(e) => setGridSize(Number(e.target.value))}
        />

        <label>
          <input
            type="checkbox"
            checked={enableBounds()}
            onChange={(e) => setEnableBounds(e.target.checked)}
          />
          Enable bounds
        </label>
      </div>
    </div>
  );
};
```

## Event Handling

```tsx
import { createDraggable, events } from '@neodrag/solid';

export const App: Component = () => {
  const [draggableRef, setDraggableRef] =
    createSignal<HTMLElement | null>(null);

  createDraggable(draggableRef, [
    events({
      onDragStart: (data) => console.log('Started:', data.offset),
      onDrag: (data) => console.log('Dragging:', data.offset),
      onDragEnd: (data) => console.log('Ended:', data.offset),
    }),
  ]);

  return (
    <div ref={setDraggableRef}>Check console while dragging</div>
  );
};
```

## Sortable lists

```tsx
import { For } from 'solid-js';
import { createSortable, createSortableItem, type SortableList } from '@neodrag/solid/sortable';

type Item = { id: string; label: string };

function SortableRow(props: { list: SortableList<Item>; item: Item }) {
  const [, itemRef] = createSortableItem(props.list, props.item.id);
  return <li ref={itemRef}>{props.item.label}</li>;
}

export const App: Component = () => {
  const [items, setItems] = createSignal<Item[]>([
    { id: 'a', label: 'Alpha' },
    { id: 'b', label: 'Beta' },
  ]);

  const { list, dropRef } = createSortable({
    items,
    keyBy: (i) => i.id,
    onReorder: (next) => setItems(next),
  });

  return (
    <ul ref={dropRef}>
      <For each={items()}>{(item) => <SortableRow list={list} item={item} />}</For>
    </ul>
  );
};
```

Low-level control: `const list = new Sortable({ items: () => items(), keyBy, onReorder })` then `createDroppable(list.container())` and `createDraggable(withSortableItemPlugins(list, key))` per chip.

## Drag Controls

```tsx
import { createDraggable, controls, ControlFrom } from '@neodrag/solid';

export const App: Component = () => {
  const [draggableRef, setDraggableRef] =
    createSignal<HTMLElement | null>(null);

  createDraggable(draggableRef, [
    controls({
      allow: ControlFrom.selector('.drag-handle'),
      block: ControlFrom.selector('.no-drag'),
    }),
  ]);

  return (
    <div ref={setDraggableRef}>
      <div class="drag-handle">🔸 Drag from here</div>
      <div>Content area</div>
      <div class="no-drag">❌ Can't drag from here</div>
    </div>
  );
};
```
