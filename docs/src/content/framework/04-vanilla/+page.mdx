## Usage

Use the shared `Neodrag` engine to bind an element. It returns a `Draggable` handle with `destroy()`:

### Basic

```typescript
import { Neodrag } from '@neodrag/vanilla';

const node = document.querySelector('#drag')!;
const drag = Neodrag.shared.draggable(node);

// later
drag.destroy();
```

### With plugins

```typescript
import { Neodrag, axis, grid } from '@neodrag/vanilla';

const node = document.querySelector('#drag')!;
const drag = Neodrag.shared.draggable(node, [axis('x'), grid([10, 10])]);
```

### Plugins defined elsewhere (TypeScript)

```typescript
import {
  Neodrag,
  axis,
  bounds,
  BoundsFrom,
  type Plugin,
} from '@neodrag/vanilla';

const plugins: Plugin[] = [axis('y'), bounds(BoundsFrom.parent())];

const drag = Neodrag.shared.draggable(
  document.querySelector('#drag')!,
  plugins,
);
```

## Reactive plugins with compartments

```typescript
import { Neodrag, axis, grid, Compartment } from '@neodrag/vanilla';

const axisCompartment = new Compartment(() => axis('x'));
const drag = Neodrag.shared.draggable(
  document.querySelector('#drag')!,
  () => [axisCompartment, grid([10, 10])],
);

axisCompartment.current = axis('y');
```

### Multiple compartments

```typescript
import {
  Neodrag,
  axis,
  bounds,
  BoundsFrom,
  grid,
  Compartment,
} from '@neodrag/vanilla';

let currentAxis = 'x';
let gridSize = 20;
let enableBounds = false;

const axisComp = new Compartment(() => axis(currentAxis));
const gridComp = new Compartment(() => grid([gridSize, gridSize]));
const boundsComp = new Compartment(() =>
  enableBounds ? bounds(BoundsFrom.parent()) : null,
);

const drag = Neodrag.shared.draggable(
  document.querySelector('#drag')!,
  () => [axisComp, gridComp, boundsComp],
);

function updateAxis(newAxis: 'x' | 'y' | 'both') {
  currentAxis = newAxis;
  axisComp.current = axis(currentAxis);
}
```

## Event handling

```typescript
import { Neodrag, events } from '@neodrag/vanilla';

Neodrag.shared.draggable(document.querySelector('#drag')!, [
  events({
    onDragStart: (data) => console.log('Started:', data.offset),
    onDrag: (data) => console.log('Dragging:', data.offset),
    onDragEnd: (data) => console.log('Ended:', data.offset),
  }),
]);
```

## Drag controls

```typescript
import { Neodrag, controls, ControlFrom } from '@neodrag/vanilla';

Neodrag.shared.draggable(document.querySelector('#drag')!, [
  controls({
    allow: ControlFrom.selector('.drag-handle'),
    block: ControlFrom.selector('.no-drag'),
  }),
]);
```

## Custom engine

Avoid multiple engines on one page unless you need isolated plugin defaults. Prefer `Neodrag.shared`:

```typescript
import { Neodrag } from '@neodrag/vanilla';

const engine = new Neodrag({
  onError: (error) => console.error(error),
});

const drag = engine.draggable(element, plugins);
engine.dispose(); // destroys all bindings on this engine
```

## Cleanup

```typescript
drag.destroy(); // single element
// or
engine.dispose(); // entire engine
```

```html
<div id="drag">Drag me!</div>

```

### CDN with plugins

```html
<div id="constrained-drag">Constrained dragging</div>
<script>
  NeoDrag.Neodrag.shared.draggable(
    document.getElementById('constrained-drag'),
    [
      NeoDrag.axis('x'),
      NeoDrag.bounds(NeoDrag.BoundsFrom.parent()),
      NeoDrag.grid([20, 20]),
    ],
  );
</script>
```

### CDN with compartments

```html
<div id="reactive-drag">Reactive dragging</div>
<button onclick="switchAxis()">Switch Axis</button>

<script>
  var currentAxis = 'x';
  var axisCompartment = new NeoDrag.Compartment(function () {
    return NeoDrag.axis(currentAxis);
  });

  NeoDrag.Neodrag.shared.draggable(
    document.getElementById('reactive-drag'),
    function () {
      return [axisCompartment];
    },
  );

  function switchAxis() {
    currentAxis = currentAxis === 'x' ? 'y' : 'x';
    axisCompartment.current = NeoDrag.axis(currentAxis);
  }
</script>
```

The global `NeoDrag` namespace re-exports plugins (`axis`, `grid`, `Compartment`, etc.) and the `Neodrag` engine class.
