import FrameworkSwitch from '$components/FrameworkSwitch.astro';

Pass **`onDragStart`**, **`onDrag`**, and **`onDragEnd`** on `new Draggable({ ... })` (Svelte) or the equivalent binding options in other frameworks. Callbacks receive **`DragEventData`**: drag **`offset`**, DOM nodes, and canonical **`input`** (`InteractionInput`) for pointer, keyboard, and programmatic drags.

```typescript
new Draggable({
  plugins: [/* ... */],
  onDragStart: (data) => console.log('Started dragging'),
  onDrag: (data) => console.log('Moving:', data.offset),
  onDragEnd: (data) => console.log('Finished dragging'),
});
```

The former `events()` plugin export was removed in v3 next — use binding options instead.

```tsx
import { useRef, useState } from 'react';
import { events, useDraggable } from '@neodrag/react';

function EventExample() {
  const elementRef = useRef<HTMLDivElement>(null);
  const [dragStatus, setDragStatus] = useState('idle');
  const [position, setPosition] = useState({ x: 0, y: 0 });

  useDraggable(elementRef, [
    events({
      onDragStart: () => {
        setDragStatus('dragging');
      },

      onDrag: (data) => {
        setPosition({ x: data.offset.x, y: data.offset.y });
      },

      onDragEnd: () => {
        setDragStatus('idle');
      },
    }),
  ]);

  return (
    <div ref={elementRef}>
      Status: {dragStatus}
      <br />
      Position: {position.x}, {position.y}
    </div>
  );
}
```

## Event Data

All event callbacks receive a `DragEventData` object with:

- `offset` — Authored position `{ x, y }` (length-aware units when using `position()`)
- `offsetPx` — Resolved pixel offset `{ x, y }`
- `rootNode` — Element the binding is attached to
- `visualNode` — Element currently being transformed (clone / sortable visual)
- `input` — **`InteractionInput`** for this frame (`pointer` | `keyboard` | `programmatic`)
- `pointer` — `{ x, y }` shorthand for `input.clientX` / `input.clientY`

For the native DOM event, branch on `input`:

```typescript
import { isPointerInput } from '@neodrag/core';

if (isPointerInput(data.input)) {
  data.input.native.preventDefault();
}
```

## Real-World Examples

### Position Tracking with Velocity

<FrameworkSwitch>
<div slot="svelte">

```svelte
<script>
  import { Draggable } from '@neodrag/svelte';
  import { events } from '@neodrag/svelte/plugins';

  let position = $state({ x: 0, y: 0 });
  let velocity = $state({ x: 0, y: 0 });
  let lastTime = 0;
  let lastPosition = { x: 0, y: 0 };

  const trackingHandler = events({
    onDrag: (data) => {
      const now = Date.now();
      const deltaTime = now - lastTime;

      if (deltaTime > 0) {
        velocity = {
          x: Math.round(
            ((data.offset.x - lastPosition.x) / deltaTime) * 1000,
          ),
          y: Math.round(
            ((data.offset.y - lastPosition.y) / deltaTime) * 1000,
          ),
        };
      }

      position = { ...data.offset };
      lastPosition = { ...data.offset };
      lastTime = now;
    },
  });

  const drag_0 = new Draggable({ plugins: [trackingHandler] });
</script>

<div {@attach drag_0.attachment}>
  Position: ({position.x}, {position.y})
  <br />
  Velocity: ({velocity.x}px/s, {velocity.y}px/s)
</div>
```

</div>

<div slot="react">

```tsx
import { useRef, useState } from 'react';
import { events, useDraggable } from '@neodrag/react';

function VelocityTracker() {
  const ref = useRef<HTMLDivElement>(null);
  const [position, setPosition] = useState({ x: 0, y: 0 });
  const [velocity, setVelocity] = useState({ x: 0, y: 0 });

  let lastTime = useRef(0);
  let lastPosition = useRef({ x: 0, y: 0 });

  useDraggable(ref, [
    events({
      onDrag: (data) => {
        const now = Date.now();
        const deltaTime = now - lastTime.current;

        if (deltaTime > 0) {
          const velocityX = Math.round(
            ((data.offset.x - lastPosition.current.x) / deltaTime) *
              1000,
          );
          const velocityY = Math.round(
            ((data.offset.y - lastPosition.current.y) / deltaTime) *
              1000,
          );
          setVelocity({ x: velocityX, y: velocityY });
        }

        setPosition({ ...data.offset });
        lastPosition.current = { ...data.offset };
        lastTime.current = now;
      },
    }),
  ]);

  return (
    <div ref={ref}>
      Position: ({position.x}, {position.y})
      <br />
      Velocity: ({velocity.x}px/s, {velocity.y}px/s)
    </div>
  );
}
```

### Progress Indicator

```typescript
// Show progress based on how far element has moved
const maxDistance = 300;

events({
  onDragStart: () => {
    progressBar.style.display = 'block';
  },

  onDrag: (data) => {
    const distance = Math.sqrt(
      data.offset.x ** 2 + data.offset.y ** 2,
    );
    const progress = Math.min((distance / maxDistance) * 100, 100);
    progressBar.style.width = `${progress}%`;
  },

  onDragEnd: () => {
    progressBar.style.display = 'none';
  },
});
```

## How It Works

The events plugin:

1. **Stores event data** in its state during each hook
2. **Schedules callbacks** using `ctx.effect.immediate()` for fast execution
3. **Provides consistent data** across all three event types
4. **Non-cancelable** - Always runs regardless of other plugin cancellations

The callbacks fire in this order:

- `onDragStart` → `onDrag` (repeatedly) → `onDragEnd`

## API Reference

```typescript
function events(options?: {
  onDragStart?: (data: DragEventData) => void;
  onDrag?: (data: DragEventData) => void;
  onDragEnd?: (data: DragEventData) => void;
}): Plugin;
```

**Options:**

- `onDragStart` - Called when dragging begins
- `onDrag` - Called repeatedly during dragging
- `onDragEnd` - Called when dragging ends

**DragEventData:**

- `offset`, `offsetPx` — drag position (authored + pixels)
- `rootNode`, `visualNode` — bound vs visual element
- `input` — `InteractionInput` (use `isPointerInput` / `isKeyboardInput` for native events)
- `pointer` — `{ x, y }` client coordinates for this frame
